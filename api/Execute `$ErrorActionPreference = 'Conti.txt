Execute `$ErrorActionPreference = 'Continue'
function Panggil($method, $url, $body) {
  Write-Output "===== $method $url ====="
  $args = @('-sS', '-D', '-', '-o', '-', '-w', "`nHTTP_CODE:%{http_code}`n", '-X', $method, $url, '-H', 'Accept: application/json')
  if ($null -ne $body) {
    $args += @('-H', 'Content-Type: application/json', '--data-binary', $body)
  }
  & curl.exe @args
  Write-Output ""
}
